#include #include #include #include #include #include // ===================================================== // RAMAH WEATHER // ESP32-C3 // ST7920 128x64 // DHT11 + WIND + FIREBASE (REALTIME FAST UPDATE) // ===================================================== // ===================================================== // WIFI // ===================================================== const char* WIFI_SSID = "naser1"; const char* WIFI_PASSWORD = "00000000"; // ===================================================== // ST7920 LCD // ===================================================== #define LCD_DATA 7 #define LCD_CS 6 #define LCD_CLK 10 U8G2_ST7920_128X64_F_SW_SPI lcd( U8G2_R0, LCD_CLK, LCD_DATA, LCD_CS, U8X8_PIN_NONE ); // ===================================================== // DHT11 // ===================================================== #define DHT_PIN 0 #define DHT_TYPE DHT11 DHT dht( DHT_PIN, DHT_TYPE ); // معايرة الحرارة const float CALIBRATION_OFFSET = 4.0; // ===================================================== // WIND SENSOR // ===================================================== #define WIND_CLK_PIN 4 #define WIND_DT_PIN 5 const int TICKS_PER_REVOLUTION = 20; // نصف القطر 2.5 سم const float RPM_TO_KMH = 0.00942; // ===================================================== // TIMING // ===================================================== // تحديث سرعة الرياح والشاشة كل 250 ملي ثانية const unsigned long WIND_UPDATE_MS = 250; // DHT11 كل ثانيتين const unsigned long DHT_UPDATE_MS = 2000; // Firebase / Web (تم تقليلها إلى 300 ملي ثانية للتحديث اللحظي) const unsigned long FIREBASE_UPDATE_MS = 300; // ===================================================== // FIREBASE // ===================================================== const char* FIREBASE_URL = "https://rumah-wind-default-rtdb.asia-southeast1.firebasedatabase.app"; // ===================================================== // GLOBAL HTTP CLIENT (KEEP-ALIVE FOR FAST SENDING) // ===================================================== WiFiClientSecure firebaseClient; HTTPClient https; bool isHttpsBegun = false; // ===================================================== // VARIABLES // ===================================================== volatile unsigned long tickCount = 0; volatile int lastCLKState = HIGH; float temperature = 0.0; float humidity = 0.0; float rpm = 0.0; float windSpeed = 0.0; bool dhtOK = false; unsigned long lastWindUpdate = 0; unsigned long lastDHTUpdate = 0; unsigned long lastFirebaseUpdate = 0; // ===================================================== // WIND INTERRUPT // ===================================================== void IRAM_ATTR encoderISR() { int clkState = digitalRead(WIND_CLK_PIN); if (clkState != lastCLKState) { if (digitalRead(WIND_DT_PIN) != clkState) { tickCount++; } lastCLKState = clkState; } } // ===================================================== // DRAW LCD // ===================================================== void drawDisplay() { lcd.clearBuffer(); // Header lcd.setFont(u8g2_font_6x10_tf); lcd.drawStr(0, 9, "RAMAH WEATHER"); lcd.drawHLine(0, 12, 128); // Temperature lcd.setFont(u8g2_font_6x10_tf); lcd.drawStr(0, 25, "Temperature:"); lcd.setFont(u8g2_font_7x13B_tf); char tempText[20]; if (dhtOK) { snprintf(tempText, sizeof(tempText), "%.1f C", temperature); } else { strcpy(tempText, "--.- C"); } lcd.drawStr(78, 25, tempText); // Humidity lcd.setFont(u8g2_font_6x10_tf); lcd.drawStr(0, 39, "Humidity:"); lcd.setFont(u8g2_font_7x13B_tf); char humidityText[20]; if (dhtOK) { snprintf(humidityText, sizeof(humidityText), "%.0f %%", humidity); } else { strcpy(humidityText, "-- %"); } lcd.drawStr(78, 39, humidityText); // Wind lcd.setFont(u8g2_font_6x10_tf); lcd.drawStr(0, 54, "Wind:"); lcd.setFont(u8g2_font_7x13B_tf); char windText[24]; snprintf(windText, sizeof(windText), "%.1f km/h", windSpeed); lcd.drawStr(55, 54, windText); // WiFi status lcd.setFont(u8g2_font_5x8_tf); if (WiFi.status() == WL_CONNECTED) { lcd.drawStr(0, 63, "WiFi OK"); } else { lcd.drawStr(0, 63, "WiFi OFF"); } lcd.sendBuffer(); } // ===================================================== // READ DHT11 // ===================================================== void readDHT() { float h = dht.readHumidity(); float t = dht.readTemperature(); if (isnan(h) || isnan(t)) { Serial.println("DHT11 ERROR"); dhtOK = false; return; } temperature = t + CALIBRATION_OFFSET; humidity = h; dhtOK = true; Serial.print("Temperature: "); Serial.print(temperature, 1); Serial.println(" C"); Serial.print("Humidity: "); Serial.print(humidity, 0); Serial.println(" %"); } // ===================================================== // CALCULATE WIND // ===================================================== void calculateWind() { unsigned long ticks; noInterrupts(); ticks = tickCount; tickCount = 0; interrupts(); float revolutions = (float)ticks / (float)TICKS_PER_REVOLUTION; rpm = revolutions * 240.0; windSpeed = rpm * RPM_TO_KMH; Serial.print("Ticks: "); Serial.print(ticks); Serial.print(" | RPM: "); Serial.print(rpm, 1); Serial.print(" | Wind Speed: "); Serial.print(windSpeed, 1); Serial.println(" km/h"); } // ===================================================== // SEND ALL DATA TO FIREBASE (FAST OPTIMIZED) // ===================================================== bool sendAllToFirebase() { if (WiFi.status() != WL_CONNECTED) { Serial.println("Firebase: WiFi OFF"); isHttpsBegun = false; return false; } String url = String(FIREBASE_URL) + "/.json"; // إعداد الاتصال لأول مرة مع تفعيل إعادة استخدام الاتصال (Keep-Alive) if (!isHttpsBegun) { firebaseClient.setInsecure(); https.setReuse(true); if (https.begin(firebaseClient, url)) { isHttpsBegun = true; } else { Serial.println("Firebase: Begin failed"); return false; } } https.addHeader("Content-Type", "application/json"); // تجهيز نص JSON String json = "{"; json += "\"windSpeed\":" + String(windSpeed, 2); json += ",\"lastSeen\":{\".sv\":\"timestamp\"}"; json += ",\"weather\":{"; json += "\"temperature\":" + String(temperature, 1); json += ",\"humidity\":" + String(humidity, 1); json += "}}"; int httpCode = https.PATCH(json); if (httpCode >= 200 && httpCode < 300) { Serial.println("Firebase: OK"); return true; } else { Serial.print("Firebase HTTP Error: "); Serial.println(httpCode); // عند حدوث خطأ نغلق الاتصال لإعادة فتحه في المحاولة القادمة https.end(); isHttpsBegun = false; return false; } } // ===================================================== // CONNECT WIFI // ===================================================== bool connectWiFi() { if (WiFi.status() == WL_CONNECTED) { return true; } Serial.println(); Serial.print("Connecting WiFi: "); Serial.println(WIFI_SSID); WiFi.mode(WIFI_STA); WiFi.begin(WIFI_SSID, WIFI_PASSWORD); unsigned long startTime = millis(); while (WiFi.status() != WL_CONNECTED) { delay(250); Serial.print("."); if (millis() - startTime >= 15000) { Serial.println(); Serial.println("WiFi connection failed"); return false; } } Serial.println(); Serial.println("WiFi connected"); Serial.print("IP: "); Serial.println(WiFi.localIP()); return true; } // ===================================================== // SETUP // ===================================================== void setup() { Serial.begin(115200); delay(1000); Serial.println(); Serial.println("================================"); Serial.println(" RAMAH WEATHER"); Serial.println("================================"); // LCD lcd.begin(); lcd.clearBuffer(); lcd.setFont(u8g2_font_6x10_tf); lcd.drawStr(25, 25, "RAMAH WEATHER"); lcd.drawStr(35, 42, "STARTING..."); lcd.sendBuffer(); // DHT dht.begin(); // WIND pinMode(WIND_CLK_PIN, INPUT_PULLUP); pinMode(WIND_DT_PIN, INPUT_PULLUP); lastCLKState = digitalRead(WIND_CLK_PIN); attachInterrupt( digitalPinToInterrupt(WIND_CLK_PIN), encoderISR, CHANGE ); // WIFI connectWiFi(); delay(1500); readDHT(); drawDisplay(); lastWindUpdate = millis(); lastDHTUpdate = millis(); lastFirebaseUpdate = millis(); Serial.println("SYSTEM READY"); } // ===================================================== // LOOP // ===================================================== void loop() { unsigned long now = millis(); // WIND + LCD if (now - lastWindUpdate >= WIND_UPDATE_MS) { lastWindUpdate = now; calculateWind(); drawDisplay(); } // DHT11 + LCD if (now - lastDHTUpdate >= DHT_UPDATE_MS) { lastDHTUpdate = now; readDHT(); drawDisplay(); } // FIREBASE / WEB (كل 300 ملي ثانية) if (now - lastFirebaseUpdate >= FIREBASE_UPDATE_MS) { lastFirebaseUpdate = now; sendAllToFirebase(); } delay(1); }
